%% --- Portada: réplica de la lámina 1 de IREM PPT Template ----------------
%%  panel logo  pos (2.89, 1.02)" tam 4.73 x 1.27"   (encima de la foto)
%%  título      pos (0.27, 2.81)" ancho 6.13"
%%  fecha       pos (0.15, 5.18)"
%%
%%  IMAGEN DE PORTADA — una sola pieza, sin recomponer.
%%
%%  Antes se recortaban foto y panel por separado y se superponían. Eso
%%  producía dos defectos: el recorte de la foto arrastraba el trozo de
%%  panel que lo solapa (la foto empieza en el 30.12% de la imagen, pero
%%  el panel se extiende sobre ella hasta el 44.72%), y el recorte del
%%  panel se cortaba antes de su curva, dejándolo con esquina cuadrada.
%%
%%  Ahora se usa la imagen oficial íntegra, a la altura de la lámina y
%%  pegada al borde derecho. IMPORTANTE: el asset ya viene recortado de
%%  sus márgenes blancos (la imagen original traía 2.07% arriba y 1.64%
%%  abajo, lo que dejaba dos franjas blancas y la foto no llegaba a los
%%  bordes). Recortada mide 1630x1115, aspecto 1.4619: a 90 mm de alto
%%  ocupa 131.57 mm, desde x = 28.43 mm, y la foto toca arriba y abajo.
%%
%%  Su tercio izquierdo es blanco y se funde con la lámina; el panel
%%  queda entre y = 9.3 y 25.2 mm con su curva intacta, y la fotografía
%%  arranca en x = 68.1 mm. De ahí el ancho de 58 mm del bloque de texto.
\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north east, inner sep=0pt] at (current page.north east)
      {\includegraphics[height=\paperheight]{portada-completa.png}};
    %% Título y subtítulo fluyen juntos desde arriba; autoría y fecha se
    %% anclan al borde inferior. Así un título o subtítulo de varias líneas
    %% nunca se encima con la autoría, que era el defecto de fijar las
    %% cuatro posiciones por separado.
    \node[anchor=north west, text width=58mm, align=flush left, inner sep=0pt]
      at ([shift={(3.2mm,-33.7mm)}]current page.north west)
      {\bfseries\color{textonegro}\fontsize{19}{24}\selectfont\inserttitle
       \ifx\insertsubtitle\empty\else
         \\[2.4mm]{\mdseries\color{iremazul}\fontsize{12}{16}\selectfont\insertsubtitle}%
       \fi};
    \node[anchor=south west, text width=58mm, align=flush left, inner sep=0pt]
      at ([shift={(1.8mm,17mm)}]current page.south west)
      {\bfseries\color{grisoscuro}\fontsize{11}{15}\selectfont\insertauthor
       \\[0.6mm]{\mdseries\color{grisclaro}\fontsize{10}{14}\selectfont\insertdate}};
  \end{tikzpicture}%
}

%% --- Lámina divisoria de sección -----------------------------------------
\newcommand{\laminaSeccion}[1]{%
  \begingroup
  \setbeamercolor{background canvas}{bg=iremazul}
  \setbeamertemplate{headline}{}
  \setbeamertemplate{footline}{}
  \begin{frame}[plain, noframenumbering]
    \vspace*{\fill}
    {\fontsize{21}{26}\selectfont\bfseries\color{white} #1\par}
    \vskip0.55em
    {\color{iremverde}\rule{2.2cm}{2.4pt}}
  \end{frame}
  \endgroup
}
\AtBeginSection[]{\laminaSeccion{\insertsectionhead}}

%% --- Cierre ---------------------------------------------------------------
%%  Va dentro de una lámina `## {.plain}`, igual que \ideaGrande.
%%  No crea su propio frame: Quarto ya tiene la lámina abierta.
\newcommand{\laminaGracias}[1][Gracias]{%
  \vspace*{\fill}
  \hspace*{6mm}{\fontsize{34}{40}\selectfont\bfseries\color{iremazul} #1\par}
  \vskip0.5em
  \hspace*{6mm}{\color{iremverde}\rule{2.4cm}{2.4pt}}
  %% La lámina de cierre es `plain` y no lleva pie, así que los logotipos
  %% se dibujan aquí a mano, en la misma posición que en el resto.
  \logosPie
}
